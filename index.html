  <script>

    // Tạo cánh hoa hồng rơi ngẫu nhiên

    const petalsContainer = document.getElementById('petals');

    const reducedMotion = window.matchMedia(

      '(prefers-reduced-motion: reduce)'

    ).matches;

 

    const petalCount = reducedMotion ? 0 : 24;

 

    for (let i = 0; i < petalCount; i++) {

      const petal = document.createElement('span');

      petal.className = 'petal';

 

      const size = 8 + Math.random() * 12;

      const duration = 12 + Math.random() * 15;

      const delay = -(Math.random() * duration);

 

      petal.style.setProperty('--left', Math.random() * 100 + '%');

      petal.style.setProperty('--size', size + 'px');

      petal.style.setProperty('--duration', duration + 's');

      petal.style.setProperty('--delay', delay + 's');

      petal.style.setProperty('--opacity', 0.3 + Math.random() * 0.55);

 

      petalsContainer.appendChild(petal);

    }

 

    // Mở lời nhắn bí mật

    const openButton = document.getElementById('openButton');

    const secretMessage = document.getElementById('secretMessage');

    const buttonText = document.getElementById('buttonText');

 

    openButton.addEventListener('click', () => {

      const isOpening = secretMessage.hidden;

 

      secretMessage.hidden = !isOpening;

      secretMessage.classList.toggle('show', isOpening);

 

      openButton.setAttribute('aria-expanded', String(isOpening));

 

      buttonText.textContent = isOpening

        ? 'Mẹ đọc lại lời nhắn nhé!'

        : 'Mẹ ơi, mở món quà này nhé!';

 

      if (isOpening) {

        // Cuộn nhẹ để mẹ đọc được lời nhắn trên điện thoại

        setTimeout(() => {

          secretMessage.scrollIntoView({

            behavior: reducedMotion ? 'auto' : 'smooth',

            block: 'nearest'

          });

        }, 100);

      }

    });
